```ts
import FETCH_ART_SRC from '../../scripts/fetch-art.mjs?raw';
import { groupByCategory, loadDecks } from '../../src/decks';

// The two source tables. Real entries are two-space-indented `name: 'CODE',`
// lines; commented-out non-entries (`// whisk: no glyph…`) don't match the
// ^\s{2}[a-z] shape.
const MAP_BLOCK = FETCH_ART_SRC.match(/const MAP = \{([\s\S]*?)\n\};/)?.[1] ?? '';
const MULBERRY_BLOCK = FETCH_ART_SRC.match(/const MULBERRY = \{([\s\S]*?)\n\};/)?.[1] ?? '';
const parseKeys = (block: string): string[] =>
  [...block.matchAll(/^\s{2}([a-z]+):\s*'/gm)].map((m) => m[1]);
const parseValues = (block: string): Map<string, string> =>
  new Map([...block.matchAll(/^\s{2}([a-z]+):\s*'([^']+)'/gm)].map((m) => [m[1], m[2]]));
const MAP_KEYS = parseKeys(MAP_BLOCK);
const MULBERRY_KEYS = parseKeys(MULBERRY_BLOCK);
const ALL_KEYS = [...MAP_KEYS, ...MULBERRY_KEYS];

// Every art basename any deck card actually references ("art/grape.svg" -> "grape").
const REFERENCED_ART = new Set(
  loadDecks()
    .flatMap((d) => d.cards)
    .filter((c) => c.img != null)
    .map((c) => String(c.img).replace(/^art\//, '').replace(/\.svg$/, '')),

describe('fetch-art MAP/MULBERRY ↔ deck img consistency', () => {
  it('parses a non-trivial MAP and MULBERRY out of scripts/fetch-art.mjs (anti-drift guard)', () => {
    // 156 OpenMoji entries at v1.7; a collapsed parse means the regex or the
    // table moved — fail loudly instead of passing on an empty key list.

    expect(MULBERRY_KEYS.length).toBeGreaterThanOrEqual(7);
    expect(new Set(MULBERRY_KEYS).size).toBe(MULBERRY_KEYS.length);
  });

  it('no word is mapped in both MAP and MULBERRY — exactly one source per word', () => {
    // fetch-art.mjs exits non-zero on overlap, but it is a dev-time tool that
    // nothing runs automatically; pin it here so the suite sees it too.
    const mulberry = new Set(MULBERRY_KEYS);
    expect(MAP_KEYS.filter((k) => mulberry.has(k))).toEqual([]);

    // A key with no referencing card is drift: fetch-art would (re)write an
    // orphan public/art SVG that ships in the PWA precache but never renders.
    // (The reverse is NOT asserted: hand-drawn placeholders like whip/chip/gate
    // are deliberately unmapped so fetch-art never clobbers them.)
    const stale = ALL_KEYS.filter((k) => !REFERENCED_ART.has(k));
    expect(stale).toEqual([]);
  });

  it('no upstream glyph is shared by two cards in the SAME category (v1.6 P1.1)', () => {
    // The jog/run collision v1.6 fixed, guarded at its source. Two keys
    // pointing at ONE upstream file produce two byte-identical SVGs, so a
    // single "shuffle all" run over their category shows the child the
    // identical picture for two different words — indistinguishable from the
    // app having lost its place. Scoped per category, matching the real
    // exposure: a shuffle pool never spans categories, and five cross-category
    // pairs (jet/plane, dish/plate, bath/tub, hut/shed, drip/wet) ship
    // deliberately. Source-prefixed so an OpenMoji hex and a Mulberry file
    // name can never be mistaken for one another.
    //
    // SCOPE — this covers exactly ONE of the two ways the collision comes
    // back: through the tables (two keys, one upstream). It joins card text ->
    // key, so it is structurally blind to pointing a card's img straight at
    // another word's existing file (`{"text": "jog", "img": "art/run.svg"}`).
    // That path is guarded in scripts/validate-decks.mjs (cardImgsByCategory),
    // which checks the rendered artifact instead. Both are needed; neither
    // subsumes the other, so don't "simplify" by deleting one.
    const sourceByKey = new Map<string, string>([
      ...[...parseValues(MAP_BLOCK)].map(([k, hex]) => [k, `openmoji:${hex}`] as const),
      ...[...parseValues(MULBERRY_BLOCK)].map(([k, file]) => [k, `mulberry:${file}`] as const),
    ]);
    const collisions: string[] = [];
    let mapped = 0;
    for (const group of groupByCategory(loadDecks())) {
      const seen = new Map<string, string>(); // upstream -> the first card that used it
      for (const deck of group.decks) {
        for (const card of deck.cards) {
          const source = sourceByKey.get(card.text);
          if (source === undefined) continue; // hand-drawn/unmapped — nothing to collide

          const first = seen.get(source);
          if (first === undefined) seen.set(source, card.text);
          else collisions.push(`${group.id}: "${card.text}" reuses "${first}"'s ${source}`);
        }
      }
    }
    expect(collisions).toEqual([]);
    // Anti-vacuous: a key/card-text mismatch (or a collapsed parse) would
    // leave `mapped` at 0 and the assertion above trivially satisfied.
    expect(mapped).toBeGreaterThan(100);
  });

    expect(MAP_KEYS).not.toContain('whisk');
    expect(MAP_KEYS).not.toContain('spider');
    expect(MAP_KEYS).not.toContain('grapes');
  });
```

```js
function remap(svg, keepHexes) {
  // Remap every #rrggbb / #rgb fill or stroke to the nearest palette token,
  // except hexes explicitly preserved via KEEP_COLORS (color-word fills).
  // Negative lookbehind: url(#abc123)-style id references are hex-SHAPED but
  // not colors — rewriting them breaks gradient/<use> refs. (rgb() colors
  // pass through unremapped — a known limitation; SVGO's convertColors turns
  // them into hex afterwards, so the palette gate still catches them.)
  const keep = new Set((keepHexes ?? []).map((h) => h.toLowerCase()));
  const named = svg.replace(NAMED_COLOR_RE, (m, prop, sep, name) => `${prop}${sep}${NAMED_COLORS[name.toLowerCase()]}`);
  return named.replace(/(?<!url\()#[0-9a-fA-F]{6}\b/g,
                     (m) => (keep.has(m.toLowerCase()) ? m : nearest(m)))
            .replace(/(?<!url\()#([0-9a-fA-F])([0-9a-fA-F])([0-9a-fA-F])\b/g,
                     (m, r, g, b) => {
                       const full = `#${r}${r}${g}${g}${b}${b}`.toLowerCase();
                       return keep.has(full) ? m : nearest(full);
                     });
}
```

3e-ii. After the `remap()` function and before `fetchOne`, insert:

```js
// SVG's default `fill` is opaque BLACK. A paint element with no fill of its
// own and no fill-bearing ancestor therefore renders #000 — off-palette, and
// invisible to validate-decks.mjs's palette gate, which only sees hex
// literals PRESENT in the file (TODOS.md P3.17; bike.svg shipped that way in
// v1.7 and was hand-fixed). Treat the implicit black exactly as remap()
// treats an explicit one — nearest palette colour, i.e. the ink — by writing
// it in. Runs AFTER SVGO so it sees the final tree (collapseGroups /
// moveGroupAttrsToElems can move a fill up or down a level). Subtrees that
// never paint directly (<defs>, <mask>, <clipPath>, …) are left alone: a
// mask's default-black child means "hide", and inking it would change that.
const PAINT_ELEMENTS = new Set(['path', 'circle', 'rect', 'ellipse', 'polygon', 'polyline']);
const NON_RENDERING = new Set([
  'defs', 'mask', 'clipPath', 'pattern', 'marker', 'symbol', 'linearGradient', 'radialGradient',
]);
const INK = PALETTE[5]; // #3d3833
function inkDefaultFills(svg) {
  const stack = []; // one entry per open ancestor: is fill already settled below it?
  let inked = 0;
  const out = svg.replace(/<(\/?)([A-Za-z][\w:-]*)([^>]*?)(\/?)>/g, (tag, close, el, attrs, selfClose) => {
    if (close) { stack.pop(); return tag; }
    const inherited = stack.length > 0 && stack[stack.length - 1];
    const own = /(?:^|\s)fill\s*=/.test(attrs) || /(?:^|\s)style\s*=\s*"[^"]*\bfill\s*:/.test(attrs);
    const settled = inherited || own || NON_RENDERING.has(el);
    let result = tag;
    if (!settled && PAINT_ELEMENTS.has(el)) {
      inked++;
      result = `<${el}${attrs} fill="${INK}"${selfClose}>`;
    }
    if (!selfClose) stack.push(settled || PAINT_ELEMENTS.has(el));
    return result;
  });
  return { svg: out, inked };
}
```

```js
async function fetchOne(name, url) {
  const res = await fetch(url, { signal: AbortSignal.timeout(30_000) });
  if (!res.ok) throw new Error(`HTTP ${res.status} for ${url}`);
  const raw = await res.text();
  const recolored = remap(raw, KEEP_COLORS.get(name));
  // preset-default does NOT strip <script> or on* handlers — and these SVGs
  // are remote content that ends up served from the site's own origin.
  const { data } = optimize(recolored, {
    multipass: true,
    plugins: [
      // removeViewBox stays OFF: the app's art must be viewBox-only (see
      // tests/unit/art-svg-sizing.test.ts), and Mulberry roots declare a
      // width/height equal to the viewBox — exactly the case where the default
      // would drop the viewBox, leaving NO size information once
      // removeDimensions runs below.
      { name: 'preset-default', params: { overrides: { removeViewBox: false } } },
      'removeScriptElement',
      // Mulberry roots carry width="850.394" height="850.394"; OpenMoji's
      // carry none, so this is a no-op for them (outputs stay byte-identical).
      'removeDimensions',
      // on*: active content. class: Mulberry's theming hooks (aac-skin-fill)
      // — dead weight, nothing here styles by class.
      { name: 'removeAttrs', params: { attrs: ['(^on.*)', 'class'] } },
    ],
  });
  const { svg, inked } = inkDefaultFills(data);
  if (inked) console.warn(`note  ${name}: ${inked} fill-less paint element(s) would have rendered black — inked ${INK}`);
  // Belt-and-braces after SVGO: refuse to write if any active-content vector
  // survived sanitization (these bytes are served from the site's own origin).
  // Internal #id references (<use xlink:href="#x">) are allowed; anything
  // external or executable fails the word loudly instead of shipping. A
  // surviving <style> block (class-based fills some Mulberry files use) is
  // refused too — its generic .st0/#Layer_1 names would collide if inlined.
  const DENYLIST = /<\s*(script|foreignObject|style|animate|animateTransform|animateMotion|set|iframe|image)\b|javascript:|(?:xlink:)?href\s*=\s*["'](?!#)/i;
  if (DENYLIST.test(svg)) throw new Error(`sanitization denylist hit — refused to write ${name}.svg`);
  // .tmp suffix (NOT .svg) so a crash mid-write can never leave a stray file
```
